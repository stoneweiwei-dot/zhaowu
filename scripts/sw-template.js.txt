const RELEASE = "__ZHAOWU_RELEASE__";
const CACHE = `zhaowu-shell-${RELEASE.slice(0, 16)}`;
const SHELL = [
  "/",
  "/manifest.webmanifest",
  "/apple-touch-icon-r113.png",
  "/brand-ui/header-gourd-wordmark-r113.png",
  "/brand-ui/logo-horizontal.svg",
  "/brand-ui/favicon.svg",
];

async function cacheFresh(path) {
  const response = await fetch(path, { cache: "no-store" });
  if (!response.ok) return;
  const cache = await caches.open(CACHE);
  await cache.put(path, response.clone());
}

self.addEventListener("install", (event) => {
  event.waitUntil(Promise.allSettled(SHELL.map((path) => cacheFresh(path))));
  self.skipWaiting();
});

self.addEventListener("activate", (event) => {
  event.waitUntil((async () => {
    const keys = await caches.keys();
    await Promise.all(
      keys
        .filter((key) => key.startsWith("zhaowu-shell-") && key !== CACHE)
        .map((key) => caches.delete(key)),
    );
    await self.clients.claim();
    const clients = await self.clients.matchAll({ type: "window", includeUncontrolled: true });
    for (const client of clients) {
      client.postMessage({ type: "ZHAOWU_RELEASE_READY", release: RELEASE });

      // iOS can keep a Home Screen web app process alive across releases. Do not
      // rely only on the page bundle understanding the update message: an older
      // client may never reload itself. A same-origin WindowClient navigation
      // makes the newly activated worker able to self-heal that stale shell.
      try {
        const target = new URL(client.url);
        if (target.origin !== self.location.origin) continue;
        // Preserve the current route so an installed PWA can self-heal even
        // when iOS resumes it on /updates, /report, /login, or another deep
        // path. Activation runs once per release, so this does not create a
        // navigation loop.
        target.searchParams.set("zw_release", RELEASE);
        await client.navigate(target.toString());
      } catch {
        // Keep activation fail-open. The page-side release checker remains a
        // second path for clients that cannot be navigated by the worker.
      }
    }
  })());
});

self.addEventListener("message", (event) => {
  if (event.data?.type === "SKIP_WAITING") self.skipWaiting();
});

self.addEventListener("fetch", (event) => {
  const request = event.request;
  if (request.method !== "GET") return;

  const url = new URL(request.url);
  if (url.origin !== self.location.origin) return;

  if (url.pathname === "/release.json") {
    event.respondWith(fetch(request, { cache: "no-store" }));
    return;
  }

  if (request.mode === "navigate") {
    event.respondWith(
      fetch(request, { cache: "no-store" })
        .then(async (response) => {
          if (response.ok) {
            const cache = await caches.open(CACHE);
            await cache.put("/", response.clone());
          }
          return response;
        })
        .catch(async () => (await caches.match("/")) || Response.error()),
    );
    return;
  }

  if (SHELL.includes(url.pathname)) {
    event.respondWith(
      fetch(request, { cache: "no-store" })
        .then(async (response) => {
          if (response.ok) {
            const cache = await caches.open(CACHE);
            await cache.put(url.pathname, response.clone());
          }
          return response;
        })
        .catch(async () => (await caches.match(url.pathname)) || Response.error()),
    );
  }
});
